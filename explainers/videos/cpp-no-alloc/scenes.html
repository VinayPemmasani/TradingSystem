Object pools and free lists
template <class T> class Pool { public: explicit Pool(std::size_t n) : slots_(n) { for (std::size_t i = 0; i + 1 < n; ++i) slots_[i].next = &slots_[i + 1]; free_ = &slots_[0]; } T* get() noexcept { T* p = free_; if (p) free_ = p->next; return p; } void put(T* p) noexcept { p->next = free_; free_ = p; } private: std::vector<T> slots_; T* free_ = nullptr;};
