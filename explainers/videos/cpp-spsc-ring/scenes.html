Cache the other side's index
alignas(64) std::atomic<size_t> head_{0}; // producer writes size_t cached_tail_ = 0; // producer's copy alignas(64) std::atomic<size_t> tail_{0}; // consumer writes size_t cached_head_ = 0; // consumer's copy
bool try_push(const T& v) { const size_t h = head_.load(relaxed); if (h - cached_tail_ == N) { cached_tail_ = tail_.load(acquire); if (h - cached_tail_ == N) return false; // full } buf_[h & kMask] = v; head_.store(h + 1, release); // publish the slot return true; }
