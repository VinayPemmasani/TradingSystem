struct Sig { // one per instrument, in an array sized at startup uint32_t pb, qb, pa, qa; // previous best: Price(4), shares int64_t ofi; double ofi_avg, imb_avg; uint64_t t_ns;}; void on_top(Sig& s, const TopOfBook& t) { // O(1) int64_t e = 0; if (t.bid_px[0] >= s.pb) e += t.bid_qty[0]; if (t.bid_px[0] <= s.pb) e -= s.qb; if (t.ask_px[0] <= s.pa) e -= t.ask_qty[0]; if (t.ask_px[0] >= s.pa) e += s.qa; s.ofi += e; /* then store the new best into s.pb..s.qa */ if (!t.bid_qty[0] || !t.ask_qty[0]) return; // one-sided uint64_t qb = t.bid_qty[0], qt = qb + t.ask_qty[0]; uint64_t micro = t.bid_px[0] + uint64_t(t.ask_px[0] - t.bid_px[0]) * qb / qt; double imb = double(qb) / qt; // ratio: double ewma(s.imb_avg, imb, t.feed_ns - s.t_ns); mark(t.instrument); // requote once, after the ring drains}
